How syncing works

  • Portfolio text is stored in Supabase; photos are stored in Cloudinary.
  • Press Save & Publish on any device. Every other device loads the new version on its next visit, and open pages refresh automatically about once a minute.
  • Sign in with your Supabase admin email and password. Only that account can publish; visitors can only read.
  • If publishing fails, your edits stay on this device and are published the next time you save.
Supabase SQL reference (one-time setup) 
create table if not exists public.portfolio_content (
  id text primary key check (id = 'main'),
  content jsonb not null,
  updated_at timestamptz not null default now()
);

alter table public.portfolio_content enable row level security;

drop policy if exists "Public can read portfolio" on public.portfolio_content;
drop policy if exists "Portfolio admin can insert" on public.portfolio_content;
drop policy if exists "Portfolio admin can update" on public.portfolio_content;

create policy "Public can read portfolio"
on public.portfolio_content for select to anon, authenticated
using (id = 'main');

create policy "Portfolio admin can insert"
on public.portfolio_content for insert to authenticated
with check (id = 'main' and auth.uid() = 'YOUR-ADMIN-USER-UUID'::uuid);

create policy "Portfolio admin can update"
on public.portfolio_content for update to authenticated
using (id = 'main' and auth.uid() = 'YOUR-ADMIN-USER-UUID'::uuid)
with check (id = 'main' and auth.uid() = 'YOUR-ADMIN-USER-UUID'::uuid);
